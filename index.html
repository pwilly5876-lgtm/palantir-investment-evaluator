    <script>
        let projectionChart = null;

        function getCurrentCAGRs() {
            return {
                bear: parseFloat(document.getElementById('cagr-bear').value) / 100,
                base: parseFloat(document.getElementById('cagr-base').value) / 100,
                bull: parseFloat(document.getElementById('cagr-bull').value) / 100,
                sp: parseFloat(document.getElementById('cagr-sp').value) / 100
            };
        }

        function formatCurrency(amount) {
            return new Intl.NumberFormat('en-US', { style: 'currency', currency: 'USD', minimumFractionDigits: 0 }).format(Math.round(amount));
        }

        function calculateFutureValue(principal, cagr, years) {
            return principal * Math.pow(1 + cagr, years);
        }

        function calculateProjections() {
            const ticker = document.getElementById('ticker').value.trim().toUpperCase() || 'LMT';
            const principal = parseFloat(document.getElementById('principal').value) || 10000;
            const cagrs = getCurrentCAGRs();

            document.getElementById('matrix-ticker').textContent = ticker;

            document.getElementById('bear-cagr-label').textContent = (cagrs.bear * 100).toFixed(1) + '% CAGR';
            document.getElementById('base-cagr-label').textContent = (cagrs.base * 100).toFixed(1) + '% CAGR';
            document.getElementById('bull-cagr-label').textContent = (cagrs.bull * 100).toFixed(1) + '% CAGR';
            document.getElementById('sp-cagr-label').textContent = (cagrs.sp * 100).toFixed(1) + '% CAGR';

            const scenarios = ['bear', 'base', 'bull', 'sp'];
            const periods = [3, 5, 10];

            scenarios.forEach(sc => {
                periods.forEach(y => {
                    const val = calculateFutureValue(principal, cagrs[sc], y);
                    const el = document.getElementById(`${sc}-${y}`);
                    if (el) el.textContent = formatCurrency(val).replace('$', '');
                });
            });

            updateProjectionChart(principal, cagrs);
        }

        function updateProjectionChart(principal, cagrs) {
            const ctx = document.getElementById('projectionChart');
            if (!ctx) return;

            const years = Array.from({ length: 11 }, (_, i) => i);
            const datasets = [
                { label: `Bear (${(cagrs.bear*100).toFixed(1)}%)`, data: years.map(y => calculateFutureValue(principal, cagrs.bear, y)), borderColor: '#f87171', tension: 0.3 },
                { label: `Base (${(cagrs.base*100).toFixed(1)}%)`, data: years.map(y => calculateFutureValue(principal, cagrs.base, y)), borderColor: '#60a5fa', tension: 0.3 },
                { label: `Bull (${(cagrs.bull*100).toFixed(1)}%)`, data: years.map(y => calculateFutureValue(principal, cagrs.bull, y)), borderColor: '#4ade80', tension: 0.3 },
                { label: `S&P (${(cagrs.sp*100).toFixed(1)}%)`, data: years.map(y => calculateFutureValue(principal, cagrs.sp, y)), borderColor: '#c084fc', tension: 0.3 }
            ];

            if (projectionChart) projectionChart.destroy();
            projectionChart = new Chart(ctx, {
                type: 'line',
                data: { labels: years, datasets },
                options: {
                    responsive: true, maintainAspectRatio: false,
                    plugins: { legend: { position: 'top', labels: { color: '#9ca3af' } } },
                    scales: {
                        x: { title: { display: true, text: 'Years', color: '#9ca3af' }, grid: { color: '#374151' } },
                        y: { title: { display: true, text: 'Value ($)', color: '#9ca3af' }, grid: { color: '#374151' }, ticks: { callback: v => '$' + (v/1000) + 'k' } }
                    }
                }
            });
        }

        // Live Data Fetch using your pre-filled API key
        async function fetchLiveMetrics() {
            const ticker = document.getElementById('ticker').value.trim().toUpperCase();
            const apiKey = document.getElementById('apiKey').value.trim();
            const tbody = document.getElementById('metrics-body');
            const dateEl = document.getElementById('metrics-date');

            tbody.innerHTML = `<tr><td colspan="3" class="p-4 text-center text-gray-400">Fetching live data for ${ticker}...</td></tr>`;

            try {
                const url = `https://financialmodelingprep.com/api/v3/key-metrics/${ticker}?apikey=${apiKey}`;
                const response = await fetch(url);
                const data = await response.json();

                if (!data || data.length === 0 || data.error) throw new Error("No data");

                const latest = data[0];

                const metrics = [
                    { name: "Price-to-Earnings (P/E)", value: latest.peRatio ? latest.peRatio.toFixed(1) + "x" : "N/A" },
                    { name: "EV / EBITDA", value: latest.enterpriseValueOverEBITDA ? latest.enterpriseValueOverEBITDA.toFixed(1) + "x" : "N/A" },
                    { name: "Operating Margin (TTM)", value: latest.operatingMarginTTM ? (latest.operatingMarginTTM * 100).toFixed(1) + "%" : "N/A" },
                    { name: "Return on Equity (ROE)", value: latest.roeTTM ? (latest.roeTTM * 100).toFixed(1) + "%" : "N/A" },
                    { name: "Free Cash Flow Yield", value: latest.freeCashFlowYieldTTM ? (latest.freeCashFlowYieldTTM * 100).toFixed(1) + "%" : "N/A" }
                ];

                tbody.innerHTML = metrics.map(m => `
                    <tr class="border-b border-gray-700">
                        <td class="p-3 font-medium text-white">${m.name}</td>
                        <td class="p-3 text-center font-mono">${m.value}</td>
                        <td class="p-3 text-gray-400">FMP API • Latest reported</td>
                    </tr>
                `).join('');

                document.getElementById('metrics-ticker').textContent = ticker;
                dateEl.textContent = `As of ${latest.date || 'latest filing'}`;

            } catch (error) {
                tbody.innerHTML = `<tr><td colspan="3" class="p-4 text-center text-red-400">Failed to load live data. Check your API key or try again.</td></tr>`;
            }
        }

        function initializeApp() {
            document.getElementById('ticker').value = 'LMT';
            document.getElementById('principal').value = '10000';

            calculateProjections();

            document.getElementById('ticker').addEventListener('input', calculateProjections);
            document.getElementById('principal').addEventListener('input', calculateProjections);
            ['cagr-bear', 'cagr-base', 'cagr-bull', 'cagr-sp'].forEach(id => {
                document.getElementById(id).addEventListener('change', calculateProjections);
            });

            // Auto-fetch live metrics on load using your key
            setTimeout(() => {
                fetchLiveMetrics();
            }, 800);
        }

        window.onload = initializeApp;
    </script>
